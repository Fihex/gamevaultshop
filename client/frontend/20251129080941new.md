package services

import (
    "fmt"
    "gamevault/database"
    "gamevault/models"
    "os"
    "path/filepath"
    "strings"
    "time"
)

// StartCleanupTask starts a background routine that runs every 24 hours
func StartCleanupTask() {

    go cleanOrphanedFiles()

    // Create a ticker that ticks every 24 hours
    ticker := time.NewTicker(24 * time.Hour)

    // Run in a background goroutine

            select {
            case <-ticker.C:
                cleanOrphanedFiles()
            }
        }
    }()
}

func cleanOrphanedFiles() {
    fmt.Println("--- [CRON] Starting Orphaned File Cleanup ---")

    // 1. Get all valid image filenames from the Database
    var games []models.Game
    // We only need the images column
    if err := database.DB.Select("images").Find(&games).Error; err != nil {
        fmt.Printf("[CRON] Error fetching games: %v\n", err)
        return
    }

    // Create a map for fast lookup: validFiles["uuid.jpg"] = true
    validFiles := make(map[string]bool)
    for _, game := range games {
        for _, imgUrl := range game.Images {
            // Database has: "/api/images/abc-123.jpg" -> We need: "abc-123.jpg"
            parts := strings.Split(imgUrl, "/")
            if len(parts) > 0 {
                filename := parts[len(parts)-1]
                validFiles[filename] = true
            }
        }
    }

    // 2. Define the uploads directory (Must match your systemd WorkingDirectory logic)
    // Since we fixed systemd, standard relative path works fine.
    uploadsDir := "uploads"

    // 3. Walk through the directory
    files, err := os.ReadDir(uploadsDir)
    if err != nil {
        fmt.Printf("[CRON] Error reading uploads dir: %v\n", err)
        return
    }

    deletedCount := 0

    for _, file := range files {
        if file.IsDir() {
            continue
        }

        filename := file.Name()

        // 4. CHECK 1: Is the file in the database?
        if _, exists := validFiles[filename]; !exists {

            // Get file info to check age
            info, err := file.Info()
            if err != nil {
                continue
            }

            // 5. CHECK 2: Safety Buffer (Is file older than 24 hours?)
            // We do NOT want to delete a file that a user is uploading RIGHT NOW
            if time.Since(info.ModTime()) > 24*time.Hour {

                // Construct full path
                fullPath := filepath.Join(uploadsDir, filename)

                fmt.Printf("[CRON] Deleting orphan: %s\n", filename)

                }
            }
        }
    }

The short answer is: **Go doesn't know (and doesn't care) that you are using Systemd.**

Go isn't talking to Systemd directly. Instead, they communicate through **Environment Variables**, which are a standard feature of the operating system (Linux).

Here is exactly how the data flows from your Service file into your Go code:

### 1. The Setup (Systemd)

When you write this in your service file:

codeIni

```
[Service]
Environment="CLEANUP_ENABLED=true"
```

You are telling Systemd: "When you start this program, please attach a sticky note to it that says CLEANUP_ENABLED=true."

### 2. The Handoff (Linux Kernel)

Systemd asks the Linux Kernel to start your binary (gamevault-api). It passes that "sticky note" (the environment variable) to the new process. This happens at the OS level, before your Go code even runs a single line.

### 3. The Pickup (Go)

When your Go application starts, it looks at its own environment (the "sticky notes" the OS gave it).

When you run this code:

codeGo

```
val := os.Getenv("CLEANUP_ENABLED")
```

Go simply asks the OS: "Do I have a variable named CLEANUP_ENABLED?"

- If you ran it manually (export CLEANUP_ENABLED=true; ./main), the OS says **"Yes, true"**.

- If you ran it via .env file (godotenv.Load()), the library injects it, and the OS says **"Yes, true"**.

- If you ran it via Systemd (Environment="CLEANUP_ENABLED=true"), the OS says **"Yes, true"**.

**Go sees the exact same result in all three cases.** It has no idea who set the variable—it only knows the variable exists.

### Why is this good?

This is excellent design because your Go code is **decoupled** from your infrastructure.

- You can run it on Docker.

- You can run it on Kubernetes.

- You can run it on Systemd.

- You can run it manually.

You never have to change your Go code. You just change how the environment variable is passed in.

import com.gamevault.entity.Game;
import com.gamevault.repository.GameRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.context.event.ApplicationReadyEvent; // <--- Import 1
import org.springframework.context.event.EventListener;             // <--- Import 2
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.io.File;
import java.nio.file.Path;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.concurrent.TimeUnit;

@Service
public class ImageCleanupService {

    private static final Logger logger = LoggerFactory.getLogger(ImageCleanupService.class);

    @Autowired
    private GameRepository gameRepository;

    @Autowired
    private FileStorageService fileStorageService;

    // Default to false (OFF) unless configured otherwise
    @Value("${app.cleanup.enabled:false}")
    private boolean isEnabled;

    // ---------------------------------------------------------
    // 1. RUN ON STARTUP (The new part)
    // ---------------------------------------------------------
    @EventListener(ApplicationReadyEvent.class)
    public void onStartup() {
        // We simply call the main logic method
        cleanupOrphanedImages();
    }

    // ---------------------------------------------------------
    // 2. RUN ON SCHEDULE (Every day at 3 AM)
    // ---------------------------------------------------------
    @Scheduled(cron = "0 0 3 * * ?")
    public void cleanupOrphanedImages() {
        // Check the toggle switch
        if (!isEnabled) {
            logger.info("--- Orphaned Image Cleanup is DISABLED in config. Skipping. ---");
            return;
        }

        logger.info("--- [CLEANUP] Starting Orphaned Image Scan ---");

        // 1. Get DB Filenames
        Set<String> validFiles = new HashSet<>();
        List<Game> games = gameRepository.findAll();

        for (Game game : games) {
            if (game.getImages() != null) {
                for (String imgUrl : game.getImages()) {
                    String filename = imgUrl.substring(imgUrl.lastIndexOf("/") + 1);
                    validFiles.add(filename);
                }
            }
        }

        // 2. Scan Directory
        Path uploadPath = fileStorageService.getFileStorageLocation();
        File folder = uploadPath.toFile();
        File[] filesOnDisk = folder.listFiles();

        if (filesOnDisk == null) return;

        int deletedCount = 0;
        long currentTime = System.currentTimeMillis();
        long twentyFourHoursInMillis = TimeUnit.HOURS.toMillis(24);

        // 3. Compare and Delete
        for (File file : filesOnDisk) {
            if (file.isFile()) {
                String filename = file.getName();

                // If not in DB
                if (!validFiles.contains(filename)) {

                    // Safety check: Is file older than 24h?
                    long fileAge = currentTime - file.lastModified();
                    if (fileAge > twentyFourHoursInMillis) {
                        try {
                            fileStorageService.deleteFile(filename);
                            logger.info("[CLEANUP] Deleted orphan: " + filename);
                            deletedCount++;
                        } catch (Exception e) {
                            logger.error("Failed to delete file: " + filename, e);
                        }
                    }
                }
            }
        }

        if (deletedCount > 0) {
            logger.info("--- [CLEANUP] Finished. Deleted " + deletedCount + " files. ---");
        } else {
            logger.info("--- [CLEANUP] Finished. No files needed deletion. ---");
        }
    }
}
