Promise.all(...):\
Old way: Upload Image A (wait) -> Upload Image B (wait) -> Upload Image C (wait) -> Save Game.\
New way: Upload A, B, C (all at once) -> Wait for slowest -> Save Game.\
This is much faster, meaning there is less time for the network to fail or the user to get impatient.\
Note on Backend (Orphaned Files)\
Even with these frontend fixes, if the user's internet cuts out exactly after the upload finishes but before the saveGame request is sent, you will still have an orphaned file on your server (a file exists in the folder, but isn't linked to any product in the database).\
This is unavoidable on the Frontend alone. To fix that completely, you would need a Backend Cron Job (scheduled task) that runs once a day (e.g., at 3 AM) and says:\
"Look at all files in the uploads folder. If a file is older than 24 hours AND it is not mentioned in the games database table, delete it."

### The Logic

1. **Run every 24 hours** (e.g., 3:00 AM).

2. **Get a list of all images** currently referenced in the Database.

3. **List all files** in the uploads/ folder.

4. **Compare:** If a file exists on disk but is NOT in the database...

5. **Safety Check:** ...AND the file is older than 24 hours (to avoid deleting files currently being uploaded), **delete it**.

### Step 1: Create the Cleanup Service (Go)

Create a new file: services/cleanup\_service.go.

You will need a library to parse the cron schedule easily, but you can also just use a Go Ticker (simpler, no external dependencies). Let's use a standard Ticker approach.

codeGo

```
package services

import (
    "fmt"
    "gamevault/database"
    "gamevault/models"
    "os"
    "path/filepath"
    "strings"
    "time"
)

// StartCleanupTask starts a background routine that runs every 24 hours
func StartCleanupTask() {
    // Run immediately on startup (optional, good for testing)
    go cleanOrphanedFiles()

    // Create a ticker that ticks every 24 hours
    ticker := time.NewTicker(24 * time.Hour)

    // Run in a background goroutine
    go func() {
        for {
            select {
            case <-ticker.C:
                cleanOrphanedFiles()
            }
        }
    }()
}

func cleanOrphanedFiles() {
    fmt.Println("--- [CRON] Starting Orphaned File Cleanup ---")

    // 1. Get all valid image filenames from the Database
    var games []models.Game
    // We only need the images column
    if err := database.DB.Select("images").Find(&games).Error; err != nil {
        fmt.Printf("[CRON] Error fetching games: %v\n", err)
        return
    }

    // Create a map for fast lookup: validFiles["uuid.jpg"] = true
    validFiles := make(map[string]bool)
    for _, game := range games {
        for _, imgUrl := range game.Images {
            // Database has: "/api/images/abc-123.jpg" -> We need: "abc-123.jpg"
            parts := strings.Split(imgUrl, "/")
            if len(parts) > 0 {
                filename := parts[len(parts)-1]
                validFiles[filename] = true
            }
        }
    }

    // 2. Define the uploads directory (Must match your systemd WorkingDirectory logic)
    // Since we fixed systemd, standard relative path works fine.
    uploadsDir := "uploads"

    // 3. Walk through the directory
    files, err := os.ReadDir(uploadsDir)
    if err != nil {
        fmt.Printf("[CRON] Error reading uploads dir: %v\n", err)
        return
    }

    deletedCount := 0

    for _, file := range files {
        if file.IsDir() {
            continue
        }

        filename := file.Name()

        // 4. CHECK 1: Is the file in the database?
        if _, exists := validFiles[filename]; !exists {

            // Get file info to check age
            info, err := file.Info()
            if err != nil {
                continue
            }

            // 5. CHECK 2: Safety Buffer (Is file older than 24 hours?)
            // We do NOT want to delete a file that a user is uploading RIGHT NOW
            if time.Since(info.ModTime()) > 24*time.Hour {

                // Construct full path
                fullPath := filepath.Join(uploadsDir, filename)

                fmt.Printf("[CRON] Deleting orphan: %s\n", filename)
                err := os.Remove(fullPath)
                if err != nil {
                    fmt.Printf("[CRON] Failed to delete %s: %v\n", filename, err)
                } else {
                    deletedCount++
                }
            }
        }
    }

    fmt.Printf("--- [CRON] Cleanup Finished. Deleted %d files. ---\n", deletedCount)
}
```

package com.gamevault.service;

import com.gamevault.entity.Game;
import com.gamevault.repository.GameRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.io.File;
import java.nio.file.Path;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.concurrent.TimeUnit;

@Service
public class ImageCleanupService {

    private static final Logger logger = LoggerFactory.getLogger(ImageCleanupService.class);

    @Autowired
    private GameRepository gameRepository;

    @Autowired
    private FileStorageService fileStorageService;

    // Run every day at 3:00 AM
    // Cron format: Seconds Minutes Hours Day Month Weekday
    @Scheduled(cron = "0 0 3 * * ?")
    public void cleanupOrphanedImages() {
        logger.info("--- Starting Orphaned Image Cleanup ---");

        // 1. Get all valid filenames from the Database
        Set<String> validFiles = new HashSet<>();
        List<Game> games = gameRepository.findAll();

        for (Game game : games) {
            if (game.getImages() != null) {
                for (String imgUrl : game.getImages()) {
                    // Url is like "/api/images/uuid.jpg". We just want "uuid.jpg"
                    String filename = imgUrl.substring(imgUrl.lastIndexOf("/") + 1);
                    validFiles.add(filename);
                }
            }
        }

        // 2. Scan the actual directory on the server
        Path uploadPath = fileStorageService.getFileStorageLocation();
        File folder = uploadPath.toFile();
        File[] filesOnDisk = folder.listFiles();

        if (filesOnDisk == null) {
            logger.warn("Upload directory is empty or does not exist.");
            return;
        }

        int deletedCount = 0;
        long currentTime = System.currentTimeMillis();
        long twentyFourHoursInMillis = TimeUnit.HOURS.toMillis(24);

        // 3. Compare Disk vs Database
        for (File file : filesOnDisk) {
            if (file.isFile()) {
                String filename = file.getName();

                // Check if file is NOT in database
                if (!validFiles.contains(filename)) {

                    // SAFETY CHECK: Only delete if file is older than 24 hours
                    // This prevents deleting a file that is currently being uploaded
                    long fileAge = currentTime - file.lastModified();

                    if (fileAge > twentyFourHoursInMillis) {
                        try {
                            // Use your existing service to delete
                            fileStorageService.deleteFile(filename);
                            logger.info("Deleted orphaned file: " + filename);
                            deletedCount++;
                        } catch (Exception e) {
                            logger.error("Failed to delete file: " + filename, e);
                        }
                    }
                }
            }
        }

        logger.info("--- Cleanup Finished. Deleted " + deletedCount + " files. ---");
    }
}

package services

import (
    "fmt"
    "gamevault/database"
    "gamevault/models"
    "os"
    "path/filepath"
    "strconv"
    "strings"
    "time"
)

// StartCleanupTask initializes the background job
func StartCleanupTask() {
    // 1. CHECK THE TOGGLE SWITCH
    // Get value from environment variable (defaulting to "true" if not set)
    envVal := os.Getenv("CLEANUP_ENABLED")

    // If the variable is set, try to parse it
    if envVal != "" {
        isEnabled, err := strconv.ParseBool(envVal)
        if err == nil && !isEnabled {
            fmt.Println("--- [CRON] Orphaned File Cleanup is DISABLED by config. ---")
            return // Exit function, do not start the ticker
        }
    }

    fmt.Println("--- [CRON] Orphaned File Cleanup Service Started (Runs every 24h) ---")

    // 2. Start the Ticker (runs every 24 hours)
    ticker := time.NewTicker(24 * time.Hour)

    // Run in background
    go func() {
        for {
            select {
            case <-ticker.C:
                cleanOrphanedFiles()
            }
        }
    }()
}

func cleanOrphanedFiles() {
    fmt.Println("--- [CRON] Scanning for orphaned files... ---")

    // Get all games, but ONLY fetch the 'images' column for performance
    var games []models.Game
    if err := database.DB.Select("images").Find(&games).Error; err != nil {
        fmt.Printf("[CRON] Error fetching games: %v\n", err)
        return
    }

    // Create a lookup map of all images currently in use
    validFiles := make(map[string]bool)
    for _, game := range games {
        for _, imgUrl := range game.Images {
            // Extract filename from URL (e.g. "/api/images/uuid.jpg" -> "uuid.jpg")
            parts := strings.Split(imgUrl, "/")
            if len(parts) > 0 {
                filename := parts[len(parts)-1]
                validFiles[filename] = true
            }
        }
    }

    // Define uploads directory
    // Note: Relies on Systemd WorkingDirectory being set correctly!
    uploadsDir := "uploads" 

    files, err := os.ReadDir(uploadsDir)
    if err != nil {
        // Silent fail if folder doesn't exist yet
        return
    }

    deletedCount := 0

    for _, file := range files {
        if file.IsDir() {
            continue
        }

        filename := file.Name()

        // If file is NOT in the database list
        if _, exists := validFiles[filename]; !exists {

            info, err := file.Info()
            if err != nil {
                continue
            }

            // SAFETY CHECK: Only delete if older than 24 hours
            if time.Since(info.ModTime()) > 24*time.Hour {

                fullPath := filepath.Join(uploadsDir, filename)

                fmt.Printf("[CRON] Deleting orphan: %s\n", filename)
                err := os.Remove(fullPath)
                if err != nil {
                    fmt.Printf("[CRON] Failed to delete: %v\n", err)
                } else {
                    deletedCount++
                }
            }
        }
    }

    if deletedCount > 0 {
        fmt.Printf("--- [CRON] Cleanup Finished. Deleted %d files. ---\n", deletedCount)
    } else {
        fmt.Println("--- [CRON] Cleanup Finished. No files needed deletion. ---")
    }
}
